# 第五部分：关键接口与数据结构（可编译级代码）

> 所有代码使用命名空间 `Game.*`，C# 9 / Unity 2021.3+。
> 依赖 `[SerializeReference]`（Unity 2019.3+）与 `JsonUtility`。

> ## ⚠ 本文档 vs 实际代码
> **实现已落地在 `Assets/Game/` 下，以那里的代码为准。** 实施过程中发现并修正了 4 处设计问题，
> 本文档相关章节已同步，此处集中列出，避免只看旧章节的人踩坑：
>
> | # | 变更 | 原因 |
> |---|---|---|
> | 1 | `Collect<T>()` 返回 `HookScope<T>`（ref struct），调用点一律写 `using var hooks = ctx.Collect<T>();` | 原设计返回共享 List，一旦在遍历中重入同类型 Collect 就会被清空。改成带 Dispose 的租借作用域，既零 GC 又不怕重入。 |
> | 2 | `IDamageHook.OnDamaged` 的参数从 `BattleUnit attacker` 改为 `in DamageInfo dmg` | 实现者需要知道伤害**类型**。否则荆棘会反弹荆棘伤害，两个带荆棘的单位互打会乒乓到 512 次守卫上限。这是被单元测试抓出来的真实缺陷。 |
> | 3 | `EffectContext` 新增 `PreviewMode` | UI 每帧调 `CanPlayCard` 做变灰判断，会走目标解析；`RandomEnemy` 解析要消耗随机流，于是随机结果随帧率漂移，破坏「同种子同结果」。预览模式下禁止消耗随机流。 |
> | 4 | 新增 `Game.UI.InputCompat` | 工程 `activeInputHandler = 1`（只启用新输入系统），旧的 `UnityEngine.Input` 与 `StandaloneInputModule` 会在运行时抛异常。 |
>
> 代码按文件组织，直接复制到对应目录即可。

## 5.4 `Game/Core/Rng.cs`

```csharp
using System;
using System.Collections.Generic;

namespace Game.Core
{
    public enum RngStream { Map, Encounter, CardDraw, Battle, EnemyAction, Reward, Shop, Event, CardEffect }

    /// <summary>
    /// 确定性随机。每条流独立推进，互不干扰——重打一场战斗不会改变后续奖励。
    /// 借鉴 Monster Train 的 RngId（18 条流）。
    /// </summary>
    [Serializable]
    public class Rng
    {
        [Serializable] public struct StreamState { public int Stream; public uint State; }

        public int MasterSeed;
        private readonly Dictionary<RngStream, uint> _states = new();

        public Rng(int masterSeed)
        {
            MasterSeed = masterSeed;
            foreach (RngStream s in Enum.GetValues(typeof(RngStream)))
                _states[s] = Hash((uint)masterSeed, (uint)s + 1u);
        }

        private static uint Hash(uint a, uint b)
        {
            unchecked { uint h = a * 2654435761u ^ b * 2246822519u; h ^= h >> 15; h *= 2246822507u; h ^= h >> 13; return h == 0 ? 1u : h; }
        }

        private uint Next(RngStream s)
        {
            unchecked
            {
                uint x = _states[s];
                x ^= x << 13; x ^= x >> 17; x ^= x << 5;   // xorshift32
                _states[s] = x;
                return x;
            }
        }

        /// <summary>[min, max) 区间整数。</summary>
        public int Range(RngStream s, int minInclusive, int maxExclusive)
        {
            if (maxExclusive <= minInclusive) return minInclusive;
            return minInclusive + (int)(Next(s) % (uint)(maxExclusive - minInclusive));
        }

        public bool Chance(RngStream s, int percent) => Range(s, 0, 100) < percent;

        public T Pick<T>(RngStream s, IReadOnlyList<T> list)
            => list == null || list.Count == 0 ? default : list[Range(s, 0, list.Count)];

        public void Shuffle<T>(RngStream s, IList<T> list)
        {
            for (int i = list.Count - 1; i > 0; i--)
            {
                int j = Range(s, 0, i + 1);
                (list[i], list[j]) = (list[j], list[i]);
            }
        }

        // ---- 存档 ----
        public List<StreamState> Save()
        {
            var l = new List<StreamState>(_states.Count);
            foreach (var kv in _states) l.Add(new StreamState { Stream = (int)kv.Key, State = kv.Value });
            return l;
        }

        public void Restore(List<StreamState> saved)
        {
            if (saved == null) return;
            foreach (var s in saved) _states[(RngStream)s.Stream] = s.State;
        }
    }
}
```

> **升级为什么用「指向另一个 SO」而不是「差量配置」？**
> 差量方案（`upgradeDeltaDamage` 之类）在效果结构变化时（升级后多一个效果）会失效，且描述生成复杂。
> 指向另一张卡：直观、可视化、可任意改动，代价是 SO 数量翻倍——但那只是资产，不是代码。Slay the Spire 本体也是这么做的。

## 5.7 `Game/Cards/CardInstance.cs`

```csharp
using Game.Battle;

namespace Game.Cards
{
    /// <summary>
    /// 运行时卡牌。可克隆、可升级、可临时改费。
    /// 借鉴 Chrono Ark 的 Skill：卡是普通 C# 对象，不是资产也不是 MonoBehaviour。
    /// </summary>
    public class CardInstance
    {
        private static int _nextUid = 1;

        public readonly int Uid;
        public CardDefinition Def { get; private set; }
        public int UpgradeLevel { get; private set; }

        /// <summary>本场战斗内的费用增量（负数为减费）。战斗结束清零。</summary>
        public int BattleCostDelta;
        /// <summary>本回合内的费用增量。回合结束清零。</summary>
        public int TurnCostDelta;
        /// <summary>本场战斗内追加的关键字（如临时获得「消耗」）。</summary>
        public CardKeyword ExtraKeywords;
        /// <summary>true 表示这张卡是战斗中临时生成的，战斗结束丢弃，不进牌库。</summary>
        public bool IsTemporary;

        public CardInstance(CardDefinition def, bool temporary = false)
        {
            Uid = _nextUid++;
            Def = def;
            IsTemporary = temporary;
        }

        public string Id => Def.Id;
        public CardType Type => Def.Type;

        public bool HasKeyword(CardKeyword k) => Def.HasKeyword(k) || (ExtraKeywords & k) != 0;

        public bool CanUpgrade => Def.UpgradedVersion != null;

        public void Upgrade()
        {
            if (!CanUpgrade) return;
            Def = Def.UpgradedVersion;
            UpgradeLevel++;
        }

        /// <summary>深拷贝（用于「复制一张牌」）。</summary>
        public CardInstance Clone(bool temporary = true)
        {
            return new CardInstance(Def, temporary)
            {
                UpgradeLevel = UpgradeLevel,
                BattleCostDelta = BattleCostDelta,
                TurnCostDelta = TurnCostDelta,
                ExtraKeywords = ExtraKeywords,
            };
        }

        /// <summary>实际费用，含状态/遗物 Hook 修正。X 费卡返回 -1。</summary>
        public int GetCost(BattleContext ctx)
        {
            if (Def.CostMode == CostMode.X) return -1;
            if (Def.CostMode == CostMode.Unplayable) return int.MaxValue;

            int cost = Def.Cost + BattleCostDelta + TurnCostDelta;
            if (ctx != null)
            {
                var hooks = ctx.Collect<ICardPlayHook>();
                for (int i = 0; i < hooks.Count; i++)
                    hooks[i].Impl.ModifyCardCost(ctx, hooks[i].Src, this, ref cost);
            }
            return cost < 0 ? 0 : cost;
        }

        public void OnBattleEnd() { BattleCostDelta = 0; TurnCostDelta = 0; ExtraKeywords = CardKeyword.None; }
        public void OnTurnEnd()   { TurnCostDelta = 0; }
    }
}
```

## 5.8 `Game/Effects/EffectValue.cs`

```csharp
using System;
using Game.Units;

namespace Game.Effects
{
    public enum ValueScale
    {
        None,                    // 固定值
        PerStatusStackOnSelf,    // 每层某状态（自己）
        PerStatusStackOnTarget,  // 每层某状态（目标）
        PerCardInHand,
        PerCardPlayedThisTurn,
        PerEnemyAlive,
        PerBlockOnSelf,
        PerMissingHpOnSelf,
        XValue,                  // X 费卡消耗掉的能量
        PerRepeatIndex           // 用于「第 N 次攻击伤害递增」
    }

    /// <summary>
    /// 可缩放数值。「动态计算数值」不需要新效果类，全部在这里解决。
    /// 灵感来自 Monster Train 的 useStatusEffectStackMultiplier，扩展为通用缩放。
    /// </summary>
    [Serializable]
    public struct EffectValue
    {
        public int Base;
        public ValueScale Scale;
        public string ScaleId;     // Scale 需要 id 时使用（状态 id）
        public int PerUnit;        // 每一个单位加多少
        public int Min;            // 结果下限（0 表示不限制）
        public int Max;            // 结果上限（0 表示不限制）

        public static EffectValue Flat(int v) => new EffectValue { Base = v, PerUnit = 0 };

        public int Evaluate(EffectContext ctx, BattleUnit target)
        {
            int units = Scale switch
            {
                ValueScale.None                   => 0,
                ValueScale.PerStatusStackOnSelf   => ctx.Source?.GetStatusStacks(ScaleId) ?? 0,
                ValueScale.PerStatusStackOnTarget => target?.GetStatusStacks(ScaleId) ?? 0,
                ValueScale.PerCardInHand          => ctx.Battle.Deck.Hand.Count,
                ValueScale.PerCardPlayedThisTurn  => ctx.Battle.CardsPlayedThisTurn,
                ValueScale.PerEnemyAlive          => ctx.Battle.AliveEnemyCount,
                ValueScale.PerBlockOnSelf         => ctx.Source?.Block ?? 0,
                ValueScale.PerMissingHpOnSelf     => ctx.Source == null ? 0 : ctx.Source.MaxHp - ctx.Source.Hp,
                ValueScale.XValue                 => ctx.XValue,
                ValueScale.PerRepeatIndex         => ctx.RepeatIndex,
                _                                 => 0
            };

            int v = Base + units * PerUnit;
            if (Min != 0 && v < Min) v = Min;
            if (Max != 0 && v > Max) v = Max;
            return v;
        }
    }
}
```

```csharp
using System;
using Game.Units;

namespace Game.Effects
{
    public enum ConditionKind
    {
        Always,
        SelfHasStatus, TargetHasStatus,
        SelfHpBelowPercent, TargetHpBelowPercent,
        HandCountAtLeast, EnergyAtLeast, TurnNumberAtLeast,
        EnemyCountAtLeast, LastCardWasAttack, IsFirstTurn
    }

    [Serializable]
    public struct EffectCondition
    {
        public ConditionKind Kind;
        public string Id;
        public int Value;
        public bool Invert;

        public bool Test(EffectContext ctx, BattleUnit target)
        {
            bool r = Kind switch
            {
                ConditionKind.Always                => true,
                ConditionKind.SelfHasStatus         => (ctx.Source?.GetStatusStacks(Id) ?? 0) >= Math.Max(1, Value),
                ConditionKind.TargetHasStatus       => (target?.GetStatusStacks(Id) ?? 0) >= Math.Max(1, Value),
                ConditionKind.SelfHpBelowPercent    => ctx.Source != null && ctx.Source.Hp * 100 < ctx.Source.MaxHp * Value,
                ConditionKind.TargetHpBelowPercent  => target != null && target.Hp * 100 < target.MaxHp * Value,
                ConditionKind.HandCountAtLeast      => ctx.Battle.Deck.Hand.Count >= Value,
                ConditionKind.EnergyAtLeast         => ctx.Battle.Energy >= Value,
                ConditionKind.TurnNumberAtLeast     => ctx.Battle.TurnNumber >= Value,
                ConditionKind.EnemyCountAtLeast     => ctx.Battle.AliveEnemyCount >= Value,
                ConditionKind.LastCardWasAttack     => ctx.Battle.LastCardTypePlayed == Cards.CardType.Attack,
                ConditionKind.IsFirstTurn           => ctx.Battle.TurnNumber == 1,
                _                                   => true
            };
            return Invert ? !r : r;
        }
    }
}
```

## 5.10 `Game/Effects/TargetSelector.cs`

```csharp
using System;
using System.Collections.Generic;
using Game.Core;
using Game.Units;

namespace Game.Effects
{
    public enum TargetKind
    {
        None,               // 不需要目标（抽牌、加能量）
        Self,
        ChosenTarget,       // 玩家/AI 指定的那个目标
        AllEnemies,
        AllAllies,
        AllUnits,
        RandomEnemy,
        LowestHpEnemy,
        HighestHpEnemy,
        PreviousTargets     // ★ 上一个效果打到的目标（Monster Train 的 targets 传递）
    }

    [Serializable]
    public struct TargetSelector
    {
        public TargetKind Kind;
        public int Count;                 // RandomEnemy 时取几个，0 视为 1
        public bool ExcludeSelf;
        public string RequireStatusId;    // 只选带某状态的目标，可空

        public static TargetSelector Chosen => new TargetSelector { Kind = TargetKind.ChosenTarget };
        public static TargetSelector SelfOnly => new TargetSelector { Kind = TargetKind.Self };
        public static TargetSelector All => new TargetSelector { Kind = TargetKind.AllEnemies };
    }

    public static class TargetResolver
    {
        /// <summary>把 selector 解析成具体单位，写进 buffer（调用方负责清空/复用，避免 GC）。</summary>
        public static void Resolve(in TargetSelector sel, EffectContext ctx, List<BattleUnit> buffer)
        {
            buffer.Clear();
            var b = ctx.Battle;

            switch (sel.Kind)
            {
                case TargetKind.None: break;

                case TargetKind.Self:
                    if (ctx.Source != null) buffer.Add(ctx.Source);
                    break;

                case TargetKind.ChosenTarget:
                    if (ctx.ChosenTarget != null && ctx.ChosenTarget.IsAlive) buffer.Add(ctx.ChosenTarget);
                    break;

                case TargetKind.AllEnemies:
                    foreach (var u in b.AllUnits) if (u.IsAlive && u.IsEnemyOf(ctx.Source)) buffer.Add(u);
                    break;

                case TargetKind.AllAllies:
                    foreach (var u in b.AllUnits) if (u.IsAlive && !u.IsEnemyOf(ctx.Source)) buffer.Add(u);
                    break;

                case TargetKind.AllUnits:
                    foreach (var u in b.AllUnits) if (u.IsAlive) buffer.Add(u);
                    break;

                case TargetKind.RandomEnemy:
                {
                    var pool = ctx.Battle.RentUnitBuffer();
                    foreach (var u in b.AllUnits) if (u.IsAlive && u.IsEnemyOf(ctx.Source)) pool.Add(u);
                    int n = sel.Count <= 0 ? 1 : sel.Count;
                    for (int i = 0; i < n && pool.Count > 0; i++)
                    {
                        int idx = b.Rng.Range(RngStream.CardEffect, 0, pool.Count);
                        buffer.Add(pool[idx]);
                        // 允许重复命中：不移除。若要不重复，取消下一行注释。
                        // pool.RemoveAt(idx);
                    }
                    ctx.Battle.ReturnUnitBuffer(pool);
                    break;
                }

                case TargetKind.LowestHpEnemy:
                case TargetKind.HighestHpEnemy:
                {
                    BattleUnit best = null;
                    foreach (var u in b.AllUnits)
                    {
                        if (!u.IsAlive || !u.IsEnemyOf(ctx.Source)) continue;
                        if (best == null) { best = u; continue; }
                        bool better = sel.Kind == TargetKind.LowestHpEnemy ? u.Hp < best.Hp : u.Hp > best.Hp;
                        if (better) best = u;
                    }
                    if (best != null) buffer.Add(best);
                    break;
                }

                case TargetKind.PreviousTargets:
                    foreach (var u in ctx.LastTargets) if (u.IsAlive) buffer.Add(u);
                    break;
            }

            if (sel.ExcludeSelf) buffer.Remove(ctx.Source);

            if (!string.IsNullOrEmpty(sel.RequireStatusId))
                for (int i = buffer.Count - 1; i >= 0; i--)
                    if (buffer[i].GetStatusStacks(sel.RequireStatusId) <= 0) buffer.RemoveAt(i);
        }
```

```csharp
    public static class EffectResolver
    {
        public const int MaxDepth = 8;

        public static bool CanApplyAll(IReadOnlyList<CardEffect> effects, EffectContext ctx)
        {
            for (int i = 0; i < effects.Count; i++)
                if (effects[i] != null && !effects[i].CanApply(ctx)) return false;
            return true;
        }

        public static void ResolveAll(IReadOnlyList<CardEffect> effects, EffectContext ctx)
        {
            if (effects == null) return;
            for (int i = 0; i < effects.Count; i++)
                Resolve(effects[i], ctx);
        }

        public static void Resolve(CardEffect effect, EffectContext ctx)
        {
            if (effect == null) return;
            if (ctx.Depth > MaxDepth)
            {
                UnityEngine.Debug.LogWarning($"[EffectResolver] 递归深度超过 {MaxDepth}，已中断：{effect.GetType().Name}");
                return;
            }

            TargetResolver.Resolve(effect.Target, ctx, ctx.Targets);
            if (!effect.CanApply(ctx)) return;

            effect.Apply(ctx);
            ctx.CarryTargetsForward();

            // 效果可能造成死亡/连锁触发，在这里统一消费，避免递归。
            ctx.Battle.RunTriggerQueue();
        }
    }
```

```csharp
using System.Collections.Generic;
using Game.Battle;
using Game.Enemies;
using Game.Statuses;

namespace Game.Units
{
    /// <summary>玩家和敌人共用一个类。区别只在 IsPlayer 和是否有 EnemyDef/Brain。</summary>
    public class BattleUnit
    {
        private static int _nextUid = 1;

        public readonly int Uid;
        public string Name;
        public bool IsPlayer;

        public int Hp;
        public int MaxHp;
        public int Block;

        public readonly List<StatusInstance> Statuses = new(8);

        // 仅敌人使用
        public EnemyDefinition EnemyDef;
        public EnemyBrain Brain;
        public Intent CurrentIntent;

        public bool IsAlive => Hp > 0;

        public BattleUnit(string name, int maxHp, bool isPlayer)
        {
            Uid = _nextUid++; Name = name; MaxHp = maxHp; Hp = maxHp; IsPlayer = isPlayer;
        }

        public bool IsEnemyOf(BattleUnit other) => other != null && other.IsPlayer != IsPlayer;

        // ---------- 状态 ----------
        public int GetStatusStacks(string id)
        {
            if (string.IsNullOrEmpty(id)) return 0;
            for (int i = 0; i < Statuses.Count; i++)
                if (Statuses[i].Def.Id == id) return Statuses[i].Stacks;
            return 0;
        }

        public StatusInstance FindStatus(string id)
        {
            for (int i = 0; i < Statuses.Count; i++)
                if (Statuses[i].Def.Id == id) return Statuses[i];
            return null;
        }

        public void AddStatus(BattleContext ctx, StatusDefinition def, int stacks, BattleUnit source)
        {
            if (def == null || stacks == 0) return;
            var inst = FindStatus(def.Id);
            if (inst == null)
            {
                inst = new StatusInstance(def, 0, source?.Uid ?? 0);
                Statuses.Add(inst);
            }
            inst.AddStacks(stacks);
            if (inst.Stacks <= 0) { Statuses.Remove(inst); ctx.Post(new BattleEvent(BattleEventType.StatusRemoved, Uid, Uid, 0, def.Id)); return; }
            ctx.Post(new BattleEvent(BattleEventType.StatusApplied, source?.Uid ?? 0, Uid, inst.Stacks, def.Id));
        }

        public void RemoveStatus(BattleContext ctx, string id)
        {
            var inst = FindStatus(id);
            if (inst == null) return;
            Statuses.Remove(inst);
            ctx.Post(new BattleEvent(BattleEventType.StatusRemoved, Uid, Uid, 0, id));
        }

        // ---------- 护甲 ----------
        public void AddBlock(BattleContext ctx, int amount)
        {
            if (amount <= 0) return;
            var hooks = ctx.Collect<IBlockHook>();
            for (int i = 0; i < hooks.Count; i++)
                if (hooks[i].Src.Owner == this) hooks[i].Impl.ModifyBlockGain(ctx, hooks[i].Src, ref amount);
            if (amount <= 0) return;
            Block += amount;
            ctx.Post(new BattleEvent(BattleEventType.BlockGained, Uid, Uid, amount));
        }

        /// <summary>回合开始时调用。「壁垒」通过 IBlockHook.ModifyBlockDecay 把 clear 置 false。</summary>
        public void DecayBlock(BattleContext ctx)
        {
            bool clear = true;
            var hooks = ctx.Collect<IBlockHook>();
            for (int i = 0; i < hooks.Count; i++)
                if (hooks[i].Src.Owner == this) hooks[i].Impl.ModifyBlockDecay(ctx, hooks[i].Src, ref clear);
            if (clear && Block > 0)
            {
                ctx.Post(new BattleEvent(BattleEventType.BlockLost, Uid, Uid, Block));
                Block = 0;
            }
        }

        // ---------- 伤害 ----------
        /// <summary>
        /// 唯一的伤害入口。管线：攻方修正 → 受方修正 → 扣护甲 → 扣血 → OnDamaged → 死亡检查。
        /// </summary>
        public int TakeDamage(BattleContext ctx, DamageInfo dmg)
        {
            if (!IsAlive) return 0;
            dmg.Attacker ??= null;

            var hooks = ctx.Collect<IDamageHook>();

            // 1) 攻击方修正（力量等）
            if (dmg.Attacker != null)
                for (int i = 0; i < hooks.Count; i++)
                    if (hooks[i].Src.Owner == dmg.Attacker)
                        hooks[i].Impl.ModifyOutgoingDamage(ctx, hooks[i].Src, this, ref dmg);

            // 2) 受击方修正（易伤等）
            for (int i = 0; i < hooks.Count; i++)
                if (hooks[i].Src.Owner == this)
                    hooks[i].Impl.ModifyIncomingDamage(ctx, hooks[i].Src, dmg.Attacker, ref dmg);

            if (dmg.Amount < 0) dmg.Amount = 0;

            // 3) 护甲
            int blocked = 0;
            int remaining = dmg.Amount;
            if (!dmg.IgnoreBlock && Block > 0)
            {
                blocked = System.Math.Min(Block, remaining);
                Block -= blocked;
                remaining -= blocked;
                if (blocked > 0) ctx.Post(new BattleEvent(BattleEventType.DamageBlocked, dmg.Attacker?.Uid ?? 0, Uid, blocked));
            }

            // 4) 扣血
            if (remaining > 0)
            {
                Hp -= remaining;
                if (Hp < 0) Hp = 0;
                ctx.Post(new BattleEvent(BattleEventType.DamageDealt, dmg.Attacker?.Uid ?? 0, Uid, remaining));

                for (int i = 0; i < hooks.Count; i++)
                    if (hooks[i].Src.Owner == this)
                        hooks[i].Impl.OnDamaged(ctx, hooks[i].Src, dmg.Attacker, remaining);
            }

            // 5) 死亡（排队，不递归）
            if (Hp <= 0) ctx.EnqueueTrigger(() => Die(ctx));
            return remaining;
        }

        public void Heal(BattleContext ctx, int amount)
        {
            if (amount <= 0 || !IsAlive) return;
            var hooks = ctx.Collect<IHealHook>();
            for (int i = 0; i < hooks.Count; i++)
                if (hooks[i].Src.Owner == this) hooks[i].Impl.ModifyHeal(ctx, hooks[i].Src, ref amount);
            if (amount <= 0) return;
            Hp = System.Math.Min(MaxHp, Hp + amount);
            ctx.Post(new BattleEvent(BattleEventType.Healed, Uid, Uid, amount));
        }

        private bool _dead;
        public void Die(BattleContext ctx)
        {
            if (_dead || Hp > 0) return;
            _dead = true;
            ctx.Post(new BattleEvent(BattleEventType.UnitDied, Uid, Uid));
            var hooks = ctx.Collect<IDeathHook>();
            for (int i = 0; i < hooks.Count; i++) hooks[i].Impl.OnUnitDied(ctx, hooks[i].Src, this);
        }
    }
}
```

```csharp
// EnemyBrain.cs
using System.Collections.Generic;
using Game.Battle;
using Game.Core;
using Game.Effects;
using Game.Units;

namespace Game.Enemies
{
    /// <summary>
    /// 敌人决策。默认实现覆盖：固定序列 / 权重随机 / 条件过滤 / 连续使用限制 / 多阶段。
    /// 需要特殊逻辑时继承并重写 ChooseAction（借鉴 Chrono Ark 的 AI.SkillSelect）。
    /// </summary>
    public class EnemyBrain
    {
        protected BattleUnit Unit;
        protected EnemyDefinition Def;
        protected readonly List<int> History = new();
        public int Phase { get; protected set; }

        public virtual void Init(BattleUnit unit, EnemyDefinition def)
        { Unit = unit; Def = def; Phase = 0; History.Clear(); }

        /// <summary>回合开始时调用，决定这回合做什么并写入 Unit.CurrentIntent。</summary>
        public void DecideIntent(BattleContext ctx)
        {
            UpdatePhase(ctx);
            int idx = ChooseAction(ctx);
            History.Add(idx);
            var action = Def.Actions[idx];
            Unit.CurrentIntent = BuildIntent(ctx, action, idx);
            ctx.Post(new BattleEvent(BattleEventType.IntentChanged, Unit.Uid, Unit.Uid, Unit.CurrentIntent.Value, action.Name));
        }

        protected virtual void UpdatePhase(BattleContext ctx)
        {
            if (Def.PhaseHpThresholds == null || Def.PhaseHpThresholds.Count == 0) return;
            int pct = Unit.MaxHp == 0 ? 100 : Unit.Hp * 100 / Unit.MaxHp;
            int p = 0;
            for (int i = 0; i < Def.PhaseHpThresholds.Count; i++)
                if (pct <= Def.PhaseHpThresholds[i]) p = i + 1;
            if (p != Phase)
            {
                Phase = p;
                OnPhaseChanged(ctx, p);
            }
        }

        protected virtual void OnPhaseChanged(BattleContext ctx, int newPhase)
            => ctx.Post(new BattleEvent(BattleEventType.Message, Unit.Uid, Unit.Uid, newPhase, "PhaseChange"));

        /// <summary>返回要执行的 Action 在 Def.Actions 中的下标。</summary>
        protected virtual int ChooseAction(BattleContext ctx)
        {
            // 1) 固定序列
            if (Def.FixedSequence != null && Def.FixedSequence.Count > 0)
            {
                int step = History.Count;
                if (step < Def.FixedSequence.Count) return Def.FixedSequence[step];
                if (Def.LoopSequence) return Def.FixedSequence[step % Def.FixedSequence.Count];
            }

            // 2) 权重随机 + 条件 + 连续限制
            var effCtx = new EffectContext { Battle = ctx, Source = Unit, ChosenTarget = ctx.Player };
            var candidates = ctx.RentIntBuffer();
            int total = 0;
            for (int i = 0; i < Def.Actions.Count; i++)
            {
                var a = Def.Actions[i];
                if (a.Weight <= 0) continue;
                if (!a.Condition.Test(effCtx, ctx.Player)) continue;
                if (a.PhaseMask != 0 && (a.PhaseMask & (1 << Phase)) == 0) continue;
                if (a.MaxConsecutive > 0 && CountTrailing(i) >= a.MaxConsecutive) continue;
                candidates.Add(i);
                total += a.Weight;
            }

            int result = 0;
            if (candidates.Count > 0 && total > 0)
            {
                int roll = ctx.Rng.Range(RngStream.EnemyAction, 0, total);
                for (int k = 0; k < candidates.Count; k++)
                {
                    roll -= Def.Actions[candidates[k]].Weight;
                    if (roll < 0) { result = candidates[k]; break; }
                }
            }
            ctx.ReturnIntBuffer(candidates);
            return result;
        }

        protected int CountTrailing(int actionIndex)
        {
            int n = 0;
            for (int i = History.Count - 1; i >= 0 && History[i] == actionIndex; i--) n++;
            return n;
        }

        protected virtual Intent BuildIntent(BattleContext ctx, EnemyAction action, int idx)
        {
            int value = 0, times = 1;
            var effCtx = new EffectContext { Battle = ctx, Source = Unit, ChosenTarget = ctx.Player };
            foreach (var e in action.Effects)
            {
                if (e is Effects.Impl.DamageEffect d)
                { value = d.PreviewDamage(effCtx, ctx.Player); times = d.Times.Evaluate(effCtx, Unit); break; }
                if (e is Effects.Impl.BlockEffect b)
                { value = b.Amount.Evaluate(effCtx, Unit); break; }
            }
            return new Intent(action.Intent, value, times, idx);
        }

        /// <summary>执行当前意图。</summary>
        public virtual void ExecuteIntent(BattleContext ctx)
        {
            var action = Def.Actions[Unit.CurrentIntent.ActionIndex];
            var effCtx = new EffectContext { Battle = ctx, Source = Unit, ChosenTarget = ctx.Player };
            EffectResolver.ResolveAll(action.Effects, effCtx);
        }
    }
}
```

## 5.16 `Game/Battle/BattleContext.cs`（★ 核心）

```csharp
using System;
using System.Collections.Generic;
using Game.Cards;
using Game.Core;
using Game.Statuses;
using Game.Units;

namespace Game.Battle
{
    /// <summary>
    /// 战斗的全部可变状态。没有任何 Unity 依赖，可以在 EditMode 测试里完整运行一场战斗。
    /// 同时承担两个职责：
    ///   1) Collect&lt;T&gt;()：拉取式 Hook 收集（借鉴 Chrono Ark 的 IReturn&lt;T&gt;，但零 GC 且有序）
    ///   2) Post()/Events：表现事件队列（借鉴 Monster Train 的 Signal，但改为队列）
    /// </summary>
    public class BattleContext
    {
        public readonly struct Hook<T> where T : class
        {
            public readonly T Impl;
            public readonly HookSource Src;
            public Hook(T impl, HookSource src) { Impl = impl; Src = src; }
        }

        // ---------- 状态 ----------
        public BattleUnit Player;
        public readonly List<BattleUnit> AllUnits = new(8);
        public DeckController Deck;
        public Rng Rng;
        public RunContext Run;

        public BattlePhase Phase = BattlePhase.None;
        public int TurnNumber;
        public int Energy;
        public int EnergyPerTurn = 3;
        public int CardsPlayedThisTurn;
        public CardType LastCardTypePlayed;
        public bool BattleEnded;
        public bool Victory;

        /// <summary>战斗中获得的遗物行为（可选模块，第 4 阶段接入）。</summary>
        public readonly List<StatusBehaviour> RelicBehaviours = new();

        // ---------- 表现事件队列 ----------
        public readonly Queue<BattleEvent> Events = new(64);
        public void Post(in BattleEvent e) => Events.Enqueue(e);

        // ---------- 触发队列（防递归） ----------
        private readonly Queue<Action> _triggerQueue = new(16);
        private bool _runningTriggers;

        public void EnqueueTrigger(Action a) => _triggerQueue.Enqueue(a);

        /// <summary>消费触发队列。重入安全：内层调用直接返回，由最外层统一跑完。</summary>
        public void RunTriggerQueue()
        {
            if (_runningTriggers) return;
            _runningTriggers = true;
            int guard = 0;
            while (_triggerQueue.Count > 0)
            {
                if (++guard > 512)
                {
                    UnityEngine.Debug.LogError("[BattleContext] 触发队列超过 512 次，疑似死循环，已中断。");
                    _triggerQueue.Clear();
                    break;
                }
                _triggerQueue.Dequeue().Invoke();
            }
            _runningTriggers = false;
        }

        // ---------- Hook 收集 ----------
        private readonly Dictionary<Type, object> _hookBuffers = new();

        /// <summary>
        /// Collect 的返回值。用 <c>using var hooks = ctx.Collect&lt;IDamageHook&gt;();</c> 使用，
        /// 离开作用域自动把内部 List 还回池子——既零 GC，又不怕在遍历中重入同类型 Collect。
        /// </summary>
        public readonly ref struct HookScope<T> where T : class, IBattleHook
        {
            private readonly BattleContext _ctx;
            private readonly List<Hook<T>> _list;
            internal HookScope(BattleContext ctx, List<Hook<T>> list) { _ctx = ctx; _list = list; }
            public int Count => _list.Count;
            public Hook<T> this[int i] => _list[i];
            public List<Hook<T>>.Enumerator GetEnumerator() => _list.GetEnumerator();
            public void Dispose() => _ctx.ReturnHookBuffer(_list);
        }

        /// <summary>
        /// 收集所有实现了 T 的行为。来源：所有存活单位身上的状态 → 遗物。
        /// 无需订阅/退订：状态消失，Hook 自然消失。
        /// </summary>
        public HookScope<T> Collect<T>() where T : class, IBattleHook
        {
            var buffer = RentHookBuffer<T>();   // 每类型一个 Stack 池，支持嵌套调用

            for (int u = 0; u < AllUnits.Count; u++)
            {
                var unit = AllUnits[u];
                if (!unit.IsAlive) continue;
                var statuses = unit.Statuses;
                for (int s = 0; s < statuses.Count; s++)
                {
                    var inst = statuses[s];
                    var behaviours = inst.Def.Behaviours;
                    for (int b = 0; b < behaviours.Count; b++)
                        if (behaviours[b] is T hook)
                            buffer.Add(new Hook<T>(hook, new HookSource(unit, inst.Stacks, inst)));
                }
            }

            for (int r = 0; r < RelicBehaviours.Count; r++)
                if (RelicBehaviours[r] is T hook)
                    buffer.Add(new Hook<T>(hook, new HookSource(Player, 1)));

            // 插入排序：稳定排序，保证同 Order 的 Hook 顺序确定（可重现）。n 很小，性能无忧。
            // ★ 不能用 List.Sort：它不稳定，同 Order 的 Hook 顺序会随内容变化，破坏确定性。
            StableSortByOrder(buffer);
            return new HookScope<T>(this, buffer);
        }

        // ---------- 查询 ----------
        public int AliveEnemyCount
        {
            get { int n = 0; for (int i = 0; i < AllUnits.Count; i++) if (AllUnits[i].IsAlive && !AllUnits[i].IsPlayer) n++; return n; }
        }

        public BattleUnit FindUnit(int uid)
        {
            for (int i = 0; i < AllUnits.Count; i++) if (AllUnits[i].Uid == uid) return AllUnits[i];
            return null;
        }

        public void GainEnergy(int amount)
        {
            var hooks = Collect<IEnergyHook>();
            for (int i = 0; i < hooks.Count; i++) hooks[i].Impl.ModifyEnergyGain(this, hooks[i].Src, ref amount);
            Energy += amount;
            if (Energy < 0) Energy = 0;
            Post(new BattleEvent(BattleEventType.EnergyChanged, 0, 0, Energy));
        }

        public void SpendEnergy(int amount)
        {
            Energy -= amount;
            if (Energy < 0) Energy = 0;
            Post(new BattleEvent(BattleEventType.EnergyChanged, 0, 0, Energy));
        }

        // ---------- 临时缓冲池（避免每帧 new List） ----------
        private readonly Stack<List<BattleUnit>> _unitPool = new();
        public List<BattleUnit> RentUnitBuffer() => _unitPool.Count > 0 ? _unitPool.Pop() : new List<BattleUnit>(8);
        public void ReturnUnitBuffer(List<BattleUnit> l) { l.Clear(); _unitPool.Push(l); }

        private readonly Stack<List<int>> _intPool = new();
        public List<int> RentIntBuffer() => _intPool.Count > 0 ? _intPool.Pop() : new List<int>(8);
        public void ReturnIntBuffer(List<int> l) { l.Clear(); _intPool.Push(l); }
    }
}
```

> **调用约定**：一律写 `using var hooks = ctx.Collect<IDamageHook>();`（或 `using (...) { }`）。
> `HookScope<T>` 是 ref struct，离开作用域自动归还 buffer，因此**允许嵌套调用**，也不会产生 GC。
> 唯一的限制是不能把 scope 存进字段或跨 await/yield 传递——ref struct 本身就禁止这么做，编译器会拦住。

```csharp
using System.Collections.Generic;
using Game.Battle;
using Game.Core;

namespace Game.Cards
{
    public class DeckController
    {
        public readonly List<CardInstance> DrawPile = new(32);
        public readonly List<CardInstance> Hand = new(12);
        public readonly List<CardInstance> DiscardPile = new(32);
        public readonly List<CardInstance> ExhaustPile = new(16);

        public int MaxHandSize = 10;
        private BattleContext _ctx;

        public void Init(BattleContext ctx, IReadOnlyList<CardInstance> deck)
        {
            _ctx = ctx;
            DrawPile.Clear(); Hand.Clear(); DiscardPile.Clear(); ExhaustPile.Clear();

            // 固有牌先进手牌，其余进抽牌堆并洗牌
            var innate = new List<CardInstance>();
            for (int i = 0; i < deck.Count; i++)
            {
                var c = deck[i];
                if (c.HasKeyword(CardKeyword.Innate)) innate.Add(c); else DrawPile.Add(c);
            }
            ctx.Rng.Shuffle(RngStream.CardDraw, DrawPile);
            for (int i = 0; i < innate.Count; i++) DrawPile.Insert(0, innate[i]);
            ctx.Post(new BattleEvent(BattleEventType.DeckShuffled));
        }

        public int Draw(int count)
        {
            int drawn = 0;
            for (int i = 0; i < count; i++) if (DrawOne()) drawn++;
            return drawn;
        }

        public bool DrawOne()
        {
            if (Hand.Count >= MaxHandSize) return false;
            if (DrawPile.Count == 0)
            {
                if (DiscardPile.Count == 0) return false;
                ReshuffleDiscardIntoDraw();
            }
            int last = DrawPile.Count - 1;
            var card = DrawPile[last];
            DrawPile.RemoveAt(last);
            Hand.Add(card);
            _ctx.Post(new BattleEvent(BattleEventType.CardDrawn, 0, card.Uid, 0, card.Id));

            var hooks = _ctx.Collect<ICardPlayHook>();
            for (int i = 0; i < hooks.Count; i++) hooks[i].Impl.OnCardDrawn(_ctx, hooks[i].Src, card);
            return true;
        }

        public void ReshuffleDiscardIntoDraw()
        {
            DrawPile.AddRange(DiscardPile);
            DiscardPile.Clear();
            _ctx.Rng.Shuffle(RngStream.CardDraw, DrawPile);
            _ctx.Post(new BattleEvent(BattleEventType.DeckShuffled));
        }

        public void Discard(CardInstance card, bool fromHand = true)
        {
            if (fromHand && !Hand.Remove(card)) return;
            DiscardPile.Add(card);
            _ctx.Post(new BattleEvent(BattleEventType.CardDiscarded, 0, card.Uid, 0, card.Id));
            var hooks = _ctx.Collect<ICardPlayHook>();
            for (int i = 0; i < hooks.Count; i++) hooks[i].Impl.OnCardDiscarded(_ctx, hooks[i].Src, card);
        }

        public void Exhaust(CardInstance card)
        {
            Hand.Remove(card); DrawPile.Remove(card); DiscardPile.Remove(card);
            ExhaustPile.Add(card);
            _ctx.Post(new BattleEvent(BattleEventType.CardExhausted, 0, card.Uid, 0, card.Id));
        }

        /// <summary>回合结束：虚无牌消耗，保留牌留下，其余弃掉。</summary>
        public void EndTurnDiscard()
        {
            for (int i = Hand.Count - 1; i >= 0; i--)
            {
                var c = Hand[i];
                c.OnTurnEnd();
                if (c.HasKeyword(CardKeyword.Ethereal)) { Exhaust(c); continue; }
                if (c.HasKeyword(CardKeyword.Retain)) continue;
                Discard(c);
            }
        }

        public void AddCard(CardInstance card, CardPile pile, bool toTop = true)
        {
            switch (pile)
            {
                case CardPile.Hand:
                    if (Hand.Count >= MaxHandSize) { DiscardPile.Add(card); break; }
                    Hand.Add(card); break;
                case CardPile.Draw:
                    if (toTop) DrawPile.Add(card);
                    else DrawPile.Insert(_ctx.Rng.Range(RngStream.CardDraw, 0, DrawPile.Count + 1), card);
                    break;
                case CardPile.Discard: DiscardPile.Add(card); break;
                case CardPile.Exhaust: ExhaustPile.Add(card); break;
            }
            _ctx.Post(new BattleEvent(BattleEventType.CardAdded, 0, card.Uid, (int)pile, card.Id));
        }
    }
}
```

```csharp
using System.Collections.Generic;
using Game.Cards;
using Game.Core;
using Game.Effects;
using Game.Enemies;
using Game.Units;
using UnityEngine;

namespace Game.Battle
{
    /// <summary>
    /// 战斗状态机。UI 只允许调用本类的 TryPlayCard / EndTurn / GetPlayableTargets。
    /// ★ 全类不含 IEnumerator，不含 Update，不引用任何 UI 类型。
    /// </summary>
    public class BattleController : MonoBehaviour
    {
        public BattleContext Ctx { get; private set; }
        public BattlePhase Phase => Ctx?.Phase ?? BattlePhase.None;

        private readonly EffectContext _playCtx = new();

        // ------------------------------------------------------------------ 开始
        public void StartBattle(RunContext run, EncounterDefinition encounter)
        {
            Ctx = new BattleContext
            {
                Rng = run.Rng,
                Run = run,
                EnergyPerTurn = run.EnergyPerTurn,
                Deck = new DeckController()
            };

            Ctx.Player = new BattleUnit("Player", run.MaxHp, true) { Hp = run.Hp };
            Ctx.AllUnits.Add(Ctx.Player);

            foreach (var e in encounter.Enemies)
            {
                int hp = Ctx.Rng.Range(RngStream.Encounter, e.MinHp, e.MaxHp + 1);
                var unit = new BattleUnit(e.DisplayName, hp, false) { EnemyDef = e };
                unit.Brain = CreateBrain(e);
                unit.Brain.Init(unit, e);
                foreach (var s in e.StartingStatuses) unit.AddStatus(Ctx, s.Status, s.Stacks, null);
                Ctx.AllUnits.Add(unit);
            }

            Ctx.Deck.Init(Ctx, run.Deck);
            Ctx.Phase = BattlePhase.BattleStart;
            Ctx.Post(new BattleEvent(BattleEventType.BattleStarted));

            var flow = Ctx.Collect<IBattleFlowHook>();
            for (int i = 0; i < flow.Count; i++) flow[i].Impl.OnBattleStart(Ctx, flow[i].Src);
            Ctx.RunTriggerQueue();

            BeginTurn();
        }

        private static EnemyBrain CreateBrain(EnemyDefinition def)
        {
            if (string.IsNullOrEmpty(def.CustomBrainType)) return new EnemyBrain();
            var t = System.Type.GetType(def.CustomBrainType);
            if (t == null) { Debug.LogWarning($"找不到 Brain 类型 {def.CustomBrainType}，使用默认。"); return new EnemyBrain(); }
            return (EnemyBrain)System.Activator.CreateInstance(t);
        }

        // ------------------------------------------------------------------ 回合
        private void BeginTurn()
        {
            Ctx.TurnNumber++;
            Ctx.Phase = BattlePhase.TurnStart;
            Ctx.CardsPlayedThisTurn = 0;

            Ctx.Player.DecayBlock(Ctx);
            Ctx.Energy = 0;
            Ctx.GainEnergy(Ctx.EnergyPerTurn);

            var hooks = Ctx.Collect<ITurnHook>();
            for (int i = 0; i < hooks.Count; i++) hooks[i].Impl.OnTurnStart(Ctx, hooks[i].Src);
            Ctx.RunTriggerQueue();

            TickStatusDecay(atTurnStart: true);

            Ctx.Deck.Draw(Ctx.Run.CardsPerTurn);

            foreach (var u in Ctx.AllUnits)
                if (u.IsAlive && !u.IsPlayer) u.Brain.DecideIntent(Ctx);

            Ctx.Phase = BattlePhase.PlayerTurn;
            Ctx.Post(new BattleEvent(BattleEventType.TurnStarted, 0, 0, Ctx.TurnNumber));
            CheckBattleEnd();
        }

        public void EndTurn()
        {
            if (Ctx.Phase != BattlePhase.PlayerTurn) return;
            Ctx.Phase = BattlePhase.TurnEnd;

            Ctx.Deck.EndTurnDiscard();

            var hooks = Ctx.Collect<ITurnHook>();
            for (int i = 0; i < hooks.Count; i++) hooks[i].Impl.OnTurnEnd(Ctx, hooks[i].Src);
            Ctx.RunTriggerQueue();
            Ctx.Post(new BattleEvent(BattleEventType.TurnEnded, 0, 0, Ctx.TurnNumber));

            if (CheckBattleEnd()) return;

            EnemyTurn();
        }

        private void EnemyTurn()
        {
            Ctx.Phase = BattlePhase.EnemyTurn;
            Ctx.Post(new BattleEvent(BattleEventType.EnemyTurnStarted));

            for (int i = 0; i < Ctx.AllUnits.Count; i++)
            {
                var u = Ctx.AllUnits[i];
                if (!u.IsAlive || u.IsPlayer) continue;
                u.DecayBlock(Ctx);
                u.Brain.ExecuteIntent(Ctx);
                Ctx.RunTriggerQueue();
                if (CheckBattleEnd()) return;
            }

            TickStatusDecay(atTurnStart: false);
            if (CheckBattleEnd()) return;

            BeginTurn();
        }

        /// <summary>处理状态层数的自然衰减。</summary>
        private void TickStatusDecay(bool atTurnStart)
        {
            for (int u = 0; u < Ctx.AllUnits.Count; u++)
            {
                var unit = Ctx.AllUnits[u];
                for (int s = unit.Statuses.Count - 1; s >= 0; s--)
                {
                    var inst = unit.Statuses[s];
                    switch (inst.Def.Decay)
                    {
                        case Statuses.StatusDecay.LoseOneAtTurnEnd when !atTurnStart:
                            inst.Stacks--; break;
                        case Statuses.StatusDecay.RemoveAtTurnEnd when !atTurnStart:
                            inst.Stacks = 0; break;
                        case Statuses.StatusDecay.LoseAllAtTurnStart when atTurnStart:
                            inst.Stacks = 0; break;
                    }
                    if (inst.Stacks <= 0)
                    {
                        unit.Statuses.RemoveAt(s);
                        Ctx.Post(new BattleEvent(BattleEventType.StatusRemoved, unit.Uid, unit.Uid, 0, inst.Def.Id));
                    }
                }
            }
        }

        // ------------------------------------------------------------------ 出牌
        public bool CanPlayCard(CardInstance card, BattleUnit target, out PlayFailReason reason)
        {
            reason = PlayFailReason.None;
            if (Ctx.Phase != BattlePhase.PlayerTurn) { reason = PlayFailReason.NotPlayerTurn; return false; }
            if (!Ctx.Deck.Hand.Contains(card))       { reason = PlayFailReason.NotInHand; return false; }
            if (card.HasKeyword(CardKeyword.Unplayable) || card.Def.CostMode == CostMode.Unplayable)
            { reason = PlayFailReason.Unplayable; return false; }

            int cost = card.GetCost(Ctx);
            if (cost >= 0 && cost > Ctx.Energy) { reason = PlayFailReason.NotEnoughEnergy; return false; }

            if (card.Def.TargetKind == CardTargetKind.SingleEnemy)
            {
                if (target == null) { reason = PlayFailReason.NeedTarget; return false; }
                if (!target.IsAlive || target.IsPlayer) { reason = PlayFailReason.InvalidTarget; return false; }
            }

            PrepareContext(card, target, cost);
            if (!EffectResolver.CanApplyAll(card.Def.Effects, _playCtx))
            { reason = PlayFailReason.EffectCannotApply; return false; }

            return true;
        }

        private void PrepareContext(CardInstance card, BattleUnit target, int cost)
        {
            _playCtx.Battle = Ctx;
            _playCtx.Source = Ctx.Player;
            _playCtx.ChosenTarget = target;
            _playCtx.Card = card;
            _playCtx.XValue = card.Def.CostMode == CostMode.X ? Ctx.Energy : 0;
            _playCtx.Depth = 0;
            _playCtx.RepeatIndex = 0;
            _playCtx.Targets.Clear();
            _playCtx.LastTargets.Clear();
        }

        public bool TryPlayCard(CardInstance card, BattleUnit target, out PlayFailReason reason)
        {
            if (!CanPlayCard(card, target, out reason)) return false;

            int cost = card.GetCost(Ctx);
            int spend = card.Def.CostMode == CostMode.X ? Ctx.Energy : cost;
            Ctx.SpendEnergy(spend);
            PrepareContext(card, target, cost);
            _playCtx.XValue = card.Def.CostMode == CostMode.X ? spend : 0;

            // 先移出手牌，避免效果里「弃掉所有手牌」把自己也弃了
            Ctx.Deck.Hand.Remove(card);
            Ctx.CardsPlayedThisTurn++;
            Ctx.LastCardTypePlayed = card.Type;
            Ctx.Post(new BattleEvent(BattleEventType.CardPlayed, Ctx.Player.Uid, card.Uid, spend, card.Id));

            var hooks = Ctx.Collect<ICardPlayHook>();
            for (int i = 0; i < hooks.Count; i++) hooks[i].Impl.OnCardPlayed(Ctx, hooks[i].Src, card);

            EffectResolver.ResolveAll(card.Def.Effects, _playCtx);
            Ctx.RunTriggerQueue();

            // 归宿
            if (card.Type == CardType.Power)          { /* 能力牌消失，不进任何堆 */ }
            else if (card.HasKeyword(CardKeyword.Exhaust)) Ctx.Deck.Exhaust(card);
            else if (card.IsTemporary)                 Ctx.Deck.Exhaust(card);
            else                                       Ctx.Deck.Discard(card, fromHand: false);

            CheckBattleEnd();
            return true;
        }

        // ------------------------------------------------------------------ 结束
        public bool CheckBattleEnd()
        {
            if (Ctx.BattleEnded) return true;

            if (!Ctx.Player.IsAlive)      { EndBattle(false); return true; }
            if (Ctx.AliveEnemyCount == 0) { EndBattle(true);  return true; }
            return false;
        }

        private void EndBattle(bool victory)
        {
            Ctx.BattleEnded = true;
            Ctx.Victory = victory;
            Ctx.Phase = victory ? BattlePhase.Victory : BattlePhase.Defeat;

            var flow = Ctx.Collect<IBattleFlowHook>();
            for (int i = 0; i < flow.Count; i++) flow[i].Impl.OnBattleEnd(Ctx, flow[i].Src, victory);

            foreach (var c in Ctx.Run.Deck) c.OnBattleEnd();
            Ctx.Run.Hp = Ctx.Player.Hp;

            Ctx.Post(new BattleEvent(BattleEventType.BattleEnded, 0, 0, victory ? 1 : 0));
        }
    }
}
```

```csharp
using System.IO;
using Game.Cards;
using Game.Save;
using UnityEngine;

namespace Game.Core
{
    public static class SaveSystem
    {
        private static string RunPath  => Path.Combine(Application.persistentDataPath, "run.json");
        private static string MetaPath => Path.Combine(Application.persistentDataPath, "meta.json");

        // ---------- Run ----------
        public static void SaveRun(RunContext run)
        {
            var s = new RunSave
            {
                Seed = run.Seed, Hp = run.Hp, MaxHp = run.MaxHp, Gold = run.Gold,
                FloorIndex = run.FloorIndex, CurrentNodeId = run.CurrentNodeId,
                EnergyPerTurn = run.EnergyPerTurn, CardsPerTurn = run.CardsPerTurn,
                RngStates = run.Rng.Save(), Map = run.Map,
            };
            s.RelicIds.AddRange(run.RelicIds);
            foreach (var c in run.Deck)
                s.Cards.Add(new CardEntry { DefId = c.Def.Id, UpgradeLevel = c.UpgradeLevel });

            File.WriteAllText(RunPath, JsonUtility.ToJson(s, true));
        }

        public static bool TryLoadRun(GameDatabase db, out RunContext run)
        {
            run = null;
            if (!File.Exists(RunPath)) return false;

            var json = File.ReadAllText(RunPath);
            var s = JsonUtility.FromJson<RunSave>(json);
            if (s == null) return false;
            if (s.Version != SaveConstants.CurrentVersion && !Migrate(ref s)) return false;

            run = new RunContext
            {
                Seed = s.Seed, Hp = s.Hp, MaxHp = s.MaxHp, Gold = s.Gold,
                FloorIndex = s.FloorIndex, CurrentNodeId = s.CurrentNodeId,
                EnergyPerTurn = s.EnergyPerTurn, CardsPerTurn = s.CardsPerTurn,
                Map = s.Map,
            };
            run.Rng = new Rng(s.Seed);
            run.Rng.Restore(s.RngStates);
            run.RelicIds.AddRange(s.RelicIds);

            foreach (var e in s.Cards)
            {
                var def = db.GetCard(e.DefId);
                if (def == null) { Debug.LogWarning($"存档中找不到卡 {e.DefId}，已跳过。"); continue; }
                var inst = new CardInstance(def);
                for (int i = 0; i < e.UpgradeLevel; i++) inst.Upgrade();
                run.Deck.Add(inst);
            }
            return true;
        }

        public static void DeleteRun() { if (File.Exists(RunPath)) File.Delete(RunPath); }

        /// <summary>版本迁移。每加一个版本就在这里加一段 if。</summary>
        private static bool Migrate(ref RunSave s)
        {
            // if (s.Version == 1) { ...; s.Version = 2; }
            if (s.Version > SaveConstants.CurrentVersion) return false;  // 老客户端读不了新存档
            s.Version = SaveConstants.CurrentVersion;
            return true;
        }

        // ---------- Meta ----------
        public static void SaveMeta(MetaSave m) => File.WriteAllText(MetaPath, JsonUtility.ToJson(m, true));

            => File.Exists(MetaPath) ? (JsonUtility.FromJson<MetaSave>(File.ReadAllText(MetaPath)) ?? new MetaSave())
```
