## 7.0 先补一个工具方法（加到 `BattleUnit`）

意图预览需要「跑一遍伤害修正但不真的打」：

```csharp
// 追加到 Game/Units/BattleUnit.cs
public static int PreviewDamage(BattleContext ctx, BattleUnit attacker, BattleUnit target, int raw)
{
    var dmg = new DamageInfo { Amount = raw, Kind = DamageKind.Attack, Attacker = attacker };
    using var hooks = ctx.Collect<IDamageHook>();
    if (attacker != null)
        for (int i = 0; i < hooks.Count; i++)
            if (hooks[i].Src.Owner == attacker) hooks[i].Impl.ModifyOutgoingDamage(ctx, hooks[i].Src, target, ref dmg);
    if (target != null)
        for (int i = 0; i < hooks.Count; i++)
            if (hooks[i].Src.Owner == target) hooks[i].Impl.ModifyIncomingDamage(ctx, hooks[i].Src, attacker, ref dmg);
    return dmg.Amount < 0 ? 0 : dmg.Amount;
}
```

### 原子效果

```csharp
// DamageEffect.cs
using System;
using Game.Battle;
using Game.Units;

namespace Game.Effects.Impl
{
    [Serializable]
    public class DamageEffect : CardEffect
    {
        public EffectValue Amount = EffectValue.Flat(6);
        [Tooltip("攻击次数，用于多段攻击")]
        public EffectValue Times = EffectValue.Flat(1);
        public DamageKind Kind = DamageKind.Attack;
        public bool IgnoreBlock;

        public override void Apply(EffectContext ctx)
        {
            int times = Math.Max(1, Times.Evaluate(ctx, ctx.Source));
            int savedRepeat = ctx.RepeatIndex;
            for (int t = 0; t < times; t++)
            {
                ctx.RepeatIndex = t;
                for (int i = 0; i < ctx.Targets.Count; i++)
                {
                    var target = ctx.Targets[i];
                    if (!target.IsAlive) continue;
                    target.TakeDamage(ctx.Battle, new DamageInfo
                    {
                        Amount = Amount.Evaluate(ctx, target),
                        Kind = Kind,
                        IgnoreBlock = IgnoreBlock,
                        Card = ctx.Card,
                        Attacker = ctx.Source
                    });
                }
            }
            ctx.RepeatIndex = savedRepeat;
        }

        /// <summary>给敌人意图 UI 用：算出实际会造成的伤害，但不真的打。</summary>
        public int PreviewDamage(EffectContext ctx, BattleUnit target)
            => BattleUnit.PreviewDamage(ctx.Battle, ctx.Source, target, Amount.Evaluate(ctx, target));

        public override string Describe(EffectContext ctx)
            => PreviewDamage(ctx, ctx.ChosenTarget).ToString();
    }
}
```

```csharp
[Serializable]
public class EnergyEffect : CardEffect
{
    [Tooltip("正数为获得能量，负数为消耗能量")]
    public EffectValue Amount = EffectValue.Flat(1);

    public EnergyEffect() { Target = new TargetSelector { Kind = TargetKind.None }; }

    public override bool CanApply(EffectContext ctx)
    {
        int a = Amount.Evaluate(ctx, ctx.Source);
        return a >= 0 || ctx.Battle.Energy >= -a;
    }

    public override void Apply(EffectContext ctx)
    {
        int a = Amount.Evaluate(ctx, ctx.Source);
        if (a >= 0) ctx.Battle.GainEnergy(a); else ctx.Battle.SpendEnergy(-a);
    }

    public override string Describe(EffectContext ctx) => Math.Abs(Amount.Evaluate(ctx, ctx.Source)).ToString();
}
```

```csharp
// DiscardEffect.cs
using Game.Cards;
using Game.Core;

[Serializable]
public class DiscardEffect : CardEffect
{
    public enum Mode { Random, All, ChooseByPlayer }
    public Mode DiscardMode = Mode.Random;
    public EffectValue Count = EffectValue.Flat(1);

    public DiscardEffect() { Target = new TargetSelector { Kind = TargetKind.None }; }

    public override void Apply(EffectContext ctx)
    {
        var deck = ctx.Battle.Deck;
        if (DiscardMode == Mode.All)
        {
            for (int i = deck.Hand.Count - 1; i >= 0; i--) deck.Discard(deck.Hand[i]);
            return;
        }
        int n = Count.Evaluate(ctx, ctx.Source);
        for (int k = 0; k < n && deck.Hand.Count > 0; k++)
        {
            // ChooseByPlayer 在第 2 阶段接入手牌选择 UI；此处先按随机处理，保证逻辑可跑。
            int idx = ctx.Battle.Rng.Range(RngStream.CardEffect, 0, deck.Hand.Count);
            deck.Discard(deck.Hand[idx]);
        }
    }
    public override string Describe(EffectContext ctx) => Count.Evaluate(ctx, ctx.Source).ToString();
}
```

```csharp
// ExhaustEffect.cs
[Serializable]
public class ExhaustEffect : CardEffect
{
    public enum Mode { RandomInHand, AllInHand, SelfCard }
    public Mode ExhaustMode = Mode.SelfCard;
    public EffectValue Count = EffectValue.Flat(1);

    public ExhaustEffect() { Target = new TargetSelector { Kind = TargetKind.None }; }

    public override void Apply(EffectContext ctx)
    {
        var deck = ctx.Battle.Deck;
        switch (ExhaustMode)
        {
            case Mode.SelfCard:
                if (ctx.Card != null) ctx.Card.ExtraKeywords |= Cards.CardKeyword.Exhaust;
                break;
            case Mode.AllInHand:
                for (int i = deck.Hand.Count - 1; i >= 0; i--) deck.Exhaust(deck.Hand[i]);
                break;
            case Mode.RandomInHand:
                int n = Count.Evaluate(ctx, ctx.Source);
                for (int k = 0; k < n && deck.Hand.Count > 0; k++)
                    deck.Exhaust(deck.Hand[ctx.Battle.Rng.Range(Core.RngStream.CardEffect, 0, deck.Hand.Count)]);
                break;
        }
    }
}
```

```csharp
// ModifyCardCostEffect.cs
using Game.Cards;

[Serializable]
public class ModifyCardCostEffect : CardEffect
{
    public enum Scope { AllInHand, RandomInHand, SelfCard }
    public Scope Where = Scope.AllInHand;
    public EffectValue Delta = EffectValue.Flat(-1);
    [Tooltip("true = 只持续本回合")]
    public bool ThisTurnOnly = true;

    public ModifyCardCostEffect() { Target = new TargetSelector { Kind = TargetKind.None }; }

    public override void Apply(EffectContext ctx)
    {
        int d = Delta.Evaluate(ctx, ctx.Source);
        var hand = ctx.Battle.Deck.Hand;
        switch (Where)
        {
            case Scope.SelfCard: if (ctx.Card != null) ApplyTo(ctx.Card, d); break;
            case Scope.AllInHand: for (int i = 0; i < hand.Count; i++) ApplyTo(hand[i], d); break;
            case Scope.RandomInHand:
                if (hand.Count > 0) ApplyTo(hand[ctx.Battle.Rng.Range(Core.RngStream.CardEffect, 0, hand.Count)], d);
                break;
        }
    }

    private void ApplyTo(CardInstance c, int d)
    { if (ThisTurnOnly) c.TurnCostDelta += d; else c.BattleCostDelta += d; }
}
```

```csharp
// RepeatEffect.cs —— 重复执行 / 连续效果
using System.Collections.Generic;

[Serializable]
public class RepeatEffect : CardEffect
{
    public EffectValue Times = EffectValue.Flat(2);
    [SerializeReference] public List<CardEffect> Effects = new();

    public RepeatEffect() { Target = new TargetSelector { Kind = TargetKind.None }; }

    public override bool CanApply(EffectContext ctx) => EffectResolver.CanApplyAll(Effects, ctx);

    public override void Apply(EffectContext ctx)
    {
        int n = Times.Evaluate(ctx, ctx.Source);
        var child = ctx.Child();
        for (int i = 0; i < n; i++)
        {
            child.RepeatIndex = i;
            EffectResolver.ResolveAll(Effects, child);
        }
    }
}
```

```csharp
// ConditionalEffect.cs —— 条件判断
using System.Collections.Generic;

[Serializable]
public class ConditionalEffect : CardEffect
{
    public EffectCondition Condition;
    [SerializeReference] public List<CardEffect> Then = new();
    [SerializeReference] public List<CardEffect> Else = new();

    public ConditionalEffect() { Target = new TargetSelector { Kind = TargetKind.None }; }

    /// <summary>条件效果本身永远「可施放」，否则会让整张卡不可打。</summary>
    public override bool CanApply(EffectContext ctx) => true;

    public override void Apply(EffectContext ctx)
    {
        var target = ctx.ChosenTarget ?? (ctx.LastTargets.Count > 0 ? ctx.LastTargets[0] : null);
        var child = ctx.Child();
        EffectResolver.ResolveAll(Condition.Test(ctx, target) ? Then : Else, child);
    }
}
```

```csharp
// RandomPickEffect.cs —— 随机效果
using System.Collections.Generic;
using Game.Core;

[Serializable]
public class RandomPickEffect : CardEffect
{
    [System.Serializable] public class Option { public int Weight = 10; [SerializeReference] public CardEffect Effect; }
    public List<Option> Options = new();
    [Tooltip("抽几个（不重复）")] public int PickCount = 1;

    public RandomPickEffect() { Target = new TargetSelector { Kind = TargetKind.None }; }

    public override void Apply(EffectContext ctx)
    {
        if (Options.Count == 0) return;
        var pool = new List<Option>(Options);
        var child = ctx.Child();
        for (int k = 0; k < PickCount && pool.Count > 0; k++)
        {
            int total = 0; foreach (var o in pool) total += o.Weight;
            int roll = ctx.Battle.Rng.Range(RngStream.CardEffect, 0, total <= 0 ? 1 : total);
            int idx = 0;
            for (; idx < pool.Count; idx++) { roll -= pool[idx].Weight; if (roll < 0) break; }
            if (idx >= pool.Count) idx = pool.Count - 1;
            EffectResolver.Resolve(pool[idx].Effect, child);
            pool.RemoveAt(idx);
        }
    }
}
```

```csharp
// DelayedEffect.cs —— 延迟执行
using System.Collections.Generic;
using Game.Battle;

[Serializable]
public class DelayedEffect : CardEffect
{
    public enum When { EndOfThisTurn, StartOfNextTurn }
    public When Timing = When.EndOfThisTurn;
    [SerializeReference] public List<CardEffect> Effects = new();

    public DelayedEffect() { Target = new TargetSelector { Kind = TargetKind.None }; }

    public override void Apply(EffectContext ctx)
    {
        // 快照当前上下文，到时候再跑
        var snapshot = ctx.Child();
        snapshot.Depth = 0;
        ctx.Battle.ScheduleDelayed(Timing, () => EffectResolver.ResolveAll(Effects, snapshot));
    }
}
```

```csharp
// 追加到 Game/Battle/BattleContext.cs
private readonly List<(DelayedEffect.When when, Action act)> _delayed = new();

public void ScheduleDelayed(DelayedEffect.When when, Action act) => _delayed.Add((when, act));

public void FireDelayed(DelayedEffect.When when)
{
    for (int i = _delayed.Count - 1; i >= 0; i--)
        if (_delayed[i].when == when) { var a = _delayed[i].act; _delayed.RemoveAt(i); a(); }
    RunTriggerQueue();
}
```
`BattleController.EndTurn()` 里在 `OnTurnEnd` 之后调 `Ctx.FireDelayed(EndOfThisTurn)`；
`BeginTurn()` 里在 `OnTurnStart` 之后调 `Ctx.FireDelayed(StartOfNextTurn)`。

```csharp
// PoisonBehaviour.cs —— 中毒：回合结束受到等于层数的无视护甲伤害，然后 -1 层
[Serializable]
public class PoisonBehaviour : StatusBehaviour, ITurnHook
{
    public override int Order => HookOrder.AddFlat;

    public void OnTurnStart(BattleContext ctx, in HookSource src) { }

    public void OnTurnEnd(BattleContext ctx, in HookSource src)

}
```

**附赠：荆棘（演示 `OnDamaged` + 触发队列）**

```csharp
[Serializable]
public class ThornsBehaviour : StatusBehaviour, IDamageHook
{
    public void ModifyOutgoingDamage(BattleContext ctx, in HookSource src, BattleUnit target, ref DamageInfo dmg) { }
    public void ModifyIncomingDamage(BattleContext ctx, in HookSource src, BattleUnit attacker, ref DamageInfo dmg) { }

    public void OnDamaged(BattleContext ctx, in HookSource src, in DamageInfo dmg, int hpLost)
    {
        // ★ 只反弹「攻击」伤害。否则荆棘会反弹荆棘伤害，两个带荆棘的单位互打会乒乓到守卫上限。
        //    这是实施阶段被单元测试 Thorns_DoesNotRecurseInfinitely 抓出来的真实缺陷。
        if (dmg.Kind != DamageKind.Attack) return;

        var attacker = dmg.Attacker;
        if (attacker == null || !attacker.IsAlive) return;

        int n = src.Stacks;
        var owner = src.Owner;
        // ★ 排队而非立即执行，否则 A 打 B、B 反伤 A → 递归
        ctx.EnqueueTrigger(() => attacker.TakeDamage(ctx, new DamageInfo
        { Amount = n, Kind = DamageKind.Thorns, IgnoreBlock = false, Attacker = owner }));
    }
}
```

| **Adrenaline 肾上腺素** | `Cost=0, Type=Skill, TargetKind=None, Keywords=Exhaust`<br>`Effects[0] = DrawEffect { Count={Base:2} }`<br>`Effects[1] = EnergyEffect { Amount={Base:1} }`<br>`Template = "获得 {1} 点能量，抽 {0} 张牌。消耗。"` |
| **Barricade 壁垒** | `Cost=3, Type=Power, TargetKind=None`<br>`Effects[0] = ApplyStatusEffect { Target=Self, Status=Status_Barricade, Stacks={Base:1} }`<br>`Template = "护甲不再于回合开始时消失。"` |

### 逐张说明

**Strike / Defend**：最基础，验证伤害与护甲管线。

**Bash**：验证 **`PreviousTargets` 目标传递**——第二个效果不重新选目标。

**Whirlwind**：验证 **X 费 + 动态次数**。
`CostMode = X` 让 `CanPlayCard` 跳过费用检查、`TryPlayCard` 花光全部能量并把 `spend` 写进 `ctx.XValue`；
`Times.Scale = XValue` 让攻击次数等于 X。
**注意**：`Target = AllEnemies` 且 `Times` 在 `DamageEffect` 内层循环，所以是「每次都打所有敌人」，与《杀戮尖塔》一致。

**Adrenaline**：验证 **一张卡多个效果 + Exhaust 关键字**。注意 `Template` 里 `{1}` 在 `{0}` 前面——模板下标对应 `Effects` 数组下标，与文案顺序无关。

**Barricade**：验证 **能力牌 + 修改核心规则**。
`Type = Power` 让卡打出后消失（不进弃牌堆）；`BarricadeBehaviour` 通过 `IBlockHook.ModifyBlockDecay` 把 `clear` 置 false。
**整个「护甲保留」机制不需要在 `BattleController` 里加任何 if。** 这是 Hook 设计的价值所在。

### 升级版怎么做

复制 `Card_Strike.asset` → `Card_Strike+.asset`，把 `Amount.Base` 改成 9，`DisplayName` 改成「打击+」，然后把 `Card_Strike.UpgradedVersion` 指向它。`CardInstance.Upgrade()` 只是换 `Def` 引用。

需要「进入二阶段时立刻全场爆发」这种特殊表现，就写一个 20 行的 Brain：

```csharp
// Game/Enemies/Impl/GuardianBrain.cs
using Game.Battle;
using Game.Effects;
using Game.Units;

namespace Game.Enemies.Impl
{
    public class GuardianBrain : EnemyBrain
    {
        protected override void OnPhaseChanged(BattleContext ctx, int newPhase)
        {
            base.OnPhaseChanged(ctx, newPhase);
            if (newPhase != 1) return;

            // 进入二阶段：清空自身负面、获得 5 点力量
            Unit.RemoveStatus(ctx, "weak");
            Unit.RemoveStatus(ctx, "vulnerable");
            var str = ctx.Run.Database.GetStatus("strength");
            Unit.AddStatus(ctx, str, 5, Unit);
            ctx.Post(new BattleEvent(BattleEventType.Message, Unit.Uid, Unit.Uid, 0, "GuardianEnrage"));
        }

        protected override int ChooseAction(BattleContext ctx)
        {
            // 二阶段：玩家护甲高于 20 时必定使用「毁灭」
            if (Phase == 1 && ctx.Player.Block >= 20) return 4;
            return base.ChooseAction(ctx);
        }
    }
}
```

**这就是 Chrono Ark 的 `AI_Boss_Reaper` 思路，但去掉了魔数下标的坑**——`Def.Actions` 的下标仍然是魔数，所以约定：**特殊 Brain 里引用的下标必须在 `EnemyDefinition` 的 Action 名字里注明**（例如 `Actions[4].Name = "毁灭"`），并在 Brain 里加常量：

```csharp
private const int ACTION_DESTROY = 4;   // 对应 Actions[4] "毁灭"
```

## 7.5 一个可以直接跑的 EditMode 测试

证明「战斗系统可无 UI 运行」：

```csharp
// Assets/Tests/EditMode/BattleSmokeTest.cs
using NUnit.Framework;
using Game.Battle;
using Game.Cards;
using Game.Core;

public class BattleSmokeTest
{
    [Test]
    public void PlayStrike_DealsSixDamage()
    {
        var run = TestFactory.MakeRun(seed: 12345);          // 8 张打击 + 4 张防御
        var ctrl = TestFactory.MakeController();
        ctrl.StartBattle(run, TestFactory.MakeEncounter("slime"));

        var enemy = ctrl.Ctx.AllUnits[1];
        int before = enemy.Hp;

        var strike = ctrl.Ctx.Deck.Hand.Find(c => c.Id == "strike");
        Assert.IsTrue(ctrl.TryPlayCard(strike, enemy, out _));

        Assert.AreEqual(before - 6, enemy.Hp);
        Assert.AreEqual(2, ctrl.Ctx.Energy);
        Assert.Contains(strike, ctrl.Ctx.Deck.DiscardPile);
    }

    [Test]
    public void Vulnerable_Increases_Damage_By_50Percent()
    {
        var run = TestFactory.MakeRun(1);
        var ctrl = TestFactory.MakeController();
        ctrl.StartBattle(run, TestFactory.MakeEncounter("slime"));
        var enemy = ctrl.Ctx.AllUnits[1];

        enemy.AddStatus(ctrl.Ctx, TestFactory.Status("vulnerable"), 2, null);
        int before = enemy.Hp;
        ctrl.TryPlayCard(TestFactory.Card(ctrl, "strike"), enemy, out _);

        Assert.AreEqual(before - 9, enemy.Hp);   // 6 * 1.5
    }
}
```

**这两个测试跑通 = 架构的核心承诺兑现。** 如果哪天它们跑不通了，说明有人把 UI 依赖塞进了逻辑层。
